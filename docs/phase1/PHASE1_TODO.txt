### 4. Password strength meter component

Create new component:
```typescript
// frontend/react-app/src/components/PasswordStrengthMeter.tsx
import { useMemo } from 'react';

interface PasswordStrengthMeterProps {
  password: string;
}

export function PasswordStrengthMeter({ password }: PasswordStrengthMeterProps) {
  const strength = useMemo(() => {
    if (!password) return { score: 0, label: '', color: '' };
    
    let score = 0;
    if (password.length >= 8) score++;
    if (password.length >= 12) score++;
    if (/[a-z]/.test(password)) score++;
    if (/[A-Z]/.test(password)) score++;
    if (/[0-9]/.test(password)) score++;
    if (/[^a-zA-Z0-9]/.test(password)) score++;
    
    if (score <= 2) return { score, label: 'Weak', color: 'text-red-500' };
    if (score <= 4) return { score, label: 'Medium', color: 'text-yellow-500' };
    return { score, label: 'Strong', color: 'text-green-500' };
  }, [password]);

  if (!password) return null;

  return (
    <div className="space-y-1">
      <div className="flex items-center gap-2">
        <div className="flex-1 h-2 bg-muted rounded-full overflow-hidden">
          <div
            className={`h-full transition-all ${
              strength.score <= 2 ? 'bg-red-500' :
              strength.score <= 4 ? 'bg-yellow-500' : 'bg-green-500'
            }`}
            style={{ width: `${(strength.score / 6) * 100}%` }}
          />
        </div>
        <span className={`text-xs font-medium ${strength.color}`}>
          {strength.label}
        </span>
      </div>
      <p className="text-xs text-muted-foreground">
        Use 8+ characters with uppercase, lowercase, numbers, and symbols
      </p>
    </div>
  );
}
```

useEffect(() => {
  if (state.status !== 'authenticated') return;

  // Check session expiry (7 days = 604800000ms)
  const SESSION_DURATION = 7 * 24 * 60 * 60 * 1000;
  const WARNING_TIME = 5 * 60 * 1000; // Warn 5 minutes before expiry

  const checkSession = () => {
    // Get token expiry from cookie (if stored) or use default
    const lastActivity = localStorage.getItem('lastActivity');
    if (!lastActivity) {
      localStorage.setItem('lastActivity', Date.now().toString());
      return;
    }

    const timeSinceActivity = Date.now() - parseInt(lastActivity);
    const timeUntilExpiry = SESSION_DURATION - timeSinceActivity;

    if (timeUntilExpiry < WARNING_TIME && timeUntilExpiry > 0) {
      // Show warning toast
      toast.warning('Your session will expire in 5 minutes. Please save your work.');
    }
  };

  const interval = setInterval(checkSession, 60000); // Check every minute
  return () => clearInterval(interval);
}, [state.status]);

## 2) End-user chat flow — exact screens & API calls

### Method 1: Standalone link (`/chat/:slug`)

```
Screen: PublicChatPage.tsx
URL: yoursite.com/chat/creator-slug

Flow:
1. Page Load:
   GET /api/public/creator/:slug
   → Returns: { creator: { displayName, avatarUrl, welcomeMessage, popularQuestions } }

2. User Types Message:
   POST /api/public/chat
   Body: { slug, message, visitorId, sessionId? }
   → Backend:
     - Creates/reuses chat_sessions
     - Saves user message to chat_messages
     - Checks priceConfig.enablePayments
     - If payment needed → Returns: { requiresPayment: true, paymentOptions, creatorId, sessionId }
     - If free → Returns: { reply, sessionId, mirrorRunId }

3. If Payment Required:
   Frontend opens PaymentPrompt modal
   → POST /api/payments/pay-per-chat/intent { creatorId, tier, sessionId }
   → Returns: { clientSecret }
   → Stripe Elements confirms payment
   → POST /api/payments/pay-per-chat/confirm { paymentIntentId, creatorId, sessionId, tier }
   → Backend:
     - Verifies payment with Stripe
     - Saves to stripe_payments (25/75 split)
     - Regenerates full reply
   → Frontend re-sends same message → Gets full answer
```

## 5) Pay-per-chat flow — final API call sequence

### Complete flow (as per code)

```
STEP 1: User sends message
POST /api/public/chat
{
  "slug": "sarah-fitness",
  "message": "Create me a meal plan",
  "visitorId": "v_123",
  "sessionId": "session_abc" (optional)
}

Backend (publicController.ts:63-183):
├─ Finds creator by slug
├─ Checks plan limit (402 if exceeded)
├─ Creates/reuses chat_sessions
├─ Saves user message
├─ Checks: priceConfig.enablePayments === true?
├─ Checks triggers:
│  ├─ sessionMessages >= 3 (free limit)
│  ├─ keywords match
│  ├─ minLength exceeded
│  └─ alwaysRequire === true
├─ If payment needed:
│  ├─ Checks DB: stripe_payments WHERE sessionId AND status='succeeded'
│  ├─ If NOT paid → Returns:
│  │  {
│  │    requiresPayment: true,
│  │    paymentOptions: { premium: { amount: 500 }, vip: { amount: 5000 } },
│  │    creatorId: "user_123",
│  │    sessionId: "session_abc",
│  │    previewReply: "This answer requires payment..."
│  │  }
│  └─ If paid → Generates full reply
└─ If free → Generates reply immediately

STEP 2: Frontend shows payment modal
PaymentPrompt.tsx opens Stripe Elements

STEP 3: User selects tier & pays
POST /api/payments/pay-per-chat/intent
{
  "creatorId": "user_123",
  "tier": "premium", // or "vip"
  "sessionId": "session_abc",
  "visitorId": "v_123"
}

Backend (payPerChatController.ts:30-67):
├─ Gets creator priceConfig
├─ Creates Stripe PaymentIntent
├─ Returns: { clientSecret }

Frontend:
├─ Stripe Elements confirms payment (client-side)
├─ On success → POST /api/payments/pay-per-chat/confirm

STEP 4: Confirm payment
POST /api/payments/pay-per-chat/confirm
{
  "paymentIntentId": "pi_xxx",
  "creatorId": "user_123",
  "sessionId": "session_abc",
  "tier": "premium"
}

Backend (payPerChatController.ts:69-191):
├─ Retrieves PaymentIntent from Stripe
├─ Verifies status === 'succeeded'
├─ Calculates splits:
│  ├─ platformFeeCents = 25% of amount
│  └─ creatorEarningsCents = 75% of amount
├─ Saves to stripe_payments:
│  {
│    creatorId, sessionId, amount,
│    platformFeeCents, creatorEarningsCents,
│    status: 'succeeded', type: 'pay_per_chat'
│  }
├─ Regenerates full reply from last user message
├─ Saves assistant reply to chat_messages
└─ Returns: { success: true, reply: "Full meal plan..." }

STEP 5: Frontend re-sends message
POST /api/public/chat (same message)
→ Backend sees stripe_payments row exists
→ Returns full reply immediately
```
